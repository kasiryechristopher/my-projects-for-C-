// Removes spaces, tabs and line breaks from both ends of the text.
std::string trim(const std::string& text)
{
    const std::string spaces = " \t\r\n";
    std::size_t start = text.find_first_not_of(spaces);
    if (start == std::string::npos)
    {
        return "";
    }
    std::size_t end = text.find_last_not_of(spaces);
    return text.substr(start, end - start + 1);
}

std::string toUpper(std::string text)
{
    for (std::size_t i = 0; i < text.size(); ++i)
    {
        text[i] = static_cast<char>(std::toupper(static_cast<unsigned char>(text[i])));
    }
    return text;
}

std::string toLower(std::string text)
{
    for (std::size_t i = 0; i < text.size(); ++i)
    {
        text[i] = static_cast<char>(std::tolower(static_cast<unsigned char>(text[i])));
    }
    return text;
}

// True when 'part' appears anywhere inside 'text', ignoring upper/lower case.
bool containsIgnoreCase(const std::string& text, const std::string& part)
{
    return toLower(text).find(toLower(part)) != std::string::npos;
}

bool equalsIgnoreCase(const std::string& first, const std::string& second)
{
    return toLower(first) == toLower(second);
}

// Breaks one record line into its fields, e.g. "a|b||c" -> {"a","b","","c"}.
std::vector<std::string> split(const std::string& line, char delimiter)
{
    std::vector<std::string> fields;
    std::string field;
    std::istringstream stream(line);

    while (std::getline(stream, field, delimiter))
    {
        fields.push_back(field);
    }
    // getline drops a final empty field (line ending with the delimiter), so add it back
    if (!line.empty() && line[line.size() - 1] == delimiter)
    {
        fields.push_back("");
    }
    return fields;
}

// Shortens long text so that it fits in a table column, e.g. "Ssemakula John..."
std::string fitWidth(const std::string& text, std::size_t width)
{
    if (text.size() <= width)
    {
        return text;
    }
    if (width <= 3)
    {
        return text.substr(0, width);
    }
    return text.substr(0, width - 3) + "...";
}

// Adds spaces on the right so text fills a fixed width (for neat log columns).
std::string padRight(const std::string& text, std::size_t width)
{
    std::string fitted = fitWidth(text, width);
    return fitted + std::string(width - fitted.size(), ' ');
}

// zeroPad(7, 4) -> "0007"
std::string zeroPad(int number, int width)
{
    std::ostringstream out;
    out << std::setw(width) << std::setfill('0') << number;
    return out.str();
}

std::string removeChar(const std::string& text, char unwanted)
{
    std::string result;
    for (std::size_t i = 0; i < text.size(); ++i)
    {
        if (text[i] != unwanted)
        {
            result += text[i];
        }
    }
    return result;
}

bool isAllDigits(const std::string& text)
{
    if (text.empty())
    {
        return false;
    }
    for (std::size_t i = 0; i < text.size(); ++i)
    {
        if (!std::isdigit(static_cast<unsigned char>(text[i])))
        {
            return false;
        }
    }
    return true;
}

// Converts text to a whole number. Rejects text such as "12abc" or "4.5".
bool parseInt(const std::string& text, int& result)
{
    std::istringstream stream(trim(text));
    int value = 0;
    char leftover = '\0';

    if (!(stream >> value))
    {
        return false;              // not a number at all, or too large
    }
    if (stream >> leftover)
    {
        return false;              // extra characters after the number
    }
    result = value;
    return true;
}

// Converts text to a decimal number. Rejects text such as "50k" or "abc".
bool parseDouble(const std::string& text, double& result)
{
    std::istringstream stream(trim(text));
    double value = 0.0;
    char leftover = '\0';

    if (!(stream >> value))
    {
        return false;
    }
    if (stream >> leftover)
    {
        return false;
    }
    if (value != value || value > 1e15 || value < -1e15)
    {
        return false;              // reject NaN and absurdly large values
    }
    result = value;
    return true;
}

// Strict conversions used when reading data files. They THROW an exception
// (std::invalid_argument) for damaged text, so the caller can catch it.
// (Written by hand instead of std::stod/std::stoi, which some older MinGW
// compilers on Windows do not provide.)
double numberOrThrow(const std::string& text)
{
    double value = 0.0;
    if (!parseDouble(text, value))
    {
        throw std::invalid_argument("Not a valid number: '" + text + "'");
    }
    return value;
}

int wholeNumberOrThrow(const std::string& text)
{
    int value = 0;
    if (!parseInt(text, value))
    {
        throw std::invalid_argument("Not a valid whole number: '" + text + "'");
    }
    return value;
}

// Uganda Shillings have no cents in daily use, so money is rounded to whole shillings.
double roundMoney(double amount)
{
    if (amount >= 0)
    {
        return std::floor(amount + 0.5);
    }
    return -std::floor(-amount + 0.5);
}

// Adds thousands separators: 1250000 -> "1,250,000"
std::string formatMoney(double amount)
{
    double rounded = roundMoney(amount);
    bool negative = rounded < 0;
    if (negative)
    {
        rounded = -rounded;
    }

    std::ostringstream raw;
    raw << std::fixed << std::setprecision(0) << rounded;
    std::string digits = raw.str();

    std::string result;
    int counter = 0;
    for (int i = static_cast<int>(digits.size()) - 1; i >= 0; --i)
    {
        result.insert(result.begin(), digits[i]);
        ++counter;
        if (counter % 3 == 0 && i > 0)
        {
            result.insert(result.begin(), ',');
        }
    }
    return negative ? "-" + result : result;
}

std::string formatUGX(double amount)
{
    return "UGX " + formatMoney(amount);
}

// Reads the computer clock safely on every compiler.
static std::tm localNow()
{
    std::time_t now = std::time(0);
    std::tm local;
#if defined(_MSC_VER)
    localtime_s(&local, &now);       // Visual Studio's safe version
#else
    local = *std::localtime(&now);
#endif
    return local;
}

std::string today()
{
    std::tm local = localNow();
    return makeDate(local.tm_year + 1900, local.tm_mon + 1, local.tm_mday);
}

std::string currentTime()
{
    std::tm local = localNow();
    std::ostringstream out;
    out << zeroPad(local.tm_hour, 2) << ":" << zeroPad(local.tm_min, 2) << ":" << zeroPad(local.tm_sec, 2);
    return out.str();
}

bool isLeapYear(int year)
{
    return (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0);
}

int daysInMonth(int year, int month)
{
    switch (month)
    {
        case 2:
            return isLeapYear(year) ? 29 : 28;
        case 4: case 6: case 9: case 11:
            return 30;
        default:
            return 31;
    }
}

// Splits "2026-10-13" into 2026, 10 and 13. Returns false for a badly formed date.
bool splitDate(const std::string& date, int& year, int& month, int& day)
{
    if (date.size() != 10 || date[4] != '-' || date[7] != '-')
    {
        return false;
    }
    std::string y = date.substr(0, 4), m = date.substr(5, 2), d = date.substr(8, 2);
    if (!isAllDigits(y) || !isAllDigits(m) || !isAllDigits(d))
    {
        return false;
    }
    return parseInt(y, year) && parseInt(m, month) && parseInt(d, day);
}

bool isValidDate(const std::string& date)
{
    int year = 0, month = 0, day = 0;
    if (!splitDate(date, year, month, day))
    {
        return false;
    }
    if (year < 1900 || year > 2100 || month < 1 || month > 12)
    {
        return false;
    }
    return day >= 1 && day <= daysInMonth(year, month);
}

std::string makeDate(int year, int month, int day)
{
    return zeroPad(year, 4) + "-" + zeroPad(month, 2) + "-" + zeroPad(day, 2);
}

// Adds (or subtracts) whole months. 2026-01-31 + 1 month -> 2026-02-28.
std::string addMonths(const std::string& date, int months)
{
    int year = 0, month = 0, day = 0;
    if (!splitDate(date, year, month, day))
    {
        return date;
    }
    int totalMonths = year * 12 + (month - 1) + months;
    int newYear = totalMonths / 12;
    int newMonth = totalMonths % 12 + 1;
    int lastDay = daysInMonth(newYear, newMonth);
    int newDay = (day > lastDay) ? lastDay : day;
    return makeDate(newYear, newMonth, newDay);
}

// Converts a calendar date into a running day number so two dates can be subtracted.
static long dayNumber(int year, int month, int day)
{
    // Standard "days from civil date" algorithm (works for any Gregorian date).
    year -= (month <= 2) ? 1 : 0;
    long era = (year >= 0 ? year : year - 399) / 400;
    long yearOfEra = year - era * 400;
    long dayOfYear = (153 * (month + (month > 2 ? -3 : 9)) + 2) / 5 + day - 1;
    long dayOfEra = yearOfEra * 365 + yearOfEra / 4 - yearOfEra / 100 + dayOfYear;
    return era * 146097 + dayOfEra - 719468;
}

long daysBetween(const std::string& fromDate, const std::string& toDate)
{
    int y1 = 0, m1 = 0, d1 = 0, y2 = 0, m2 = 0, d2 = 0;
    if (!splitDate(fromDate, y1, m1, d1) || !splitDate(toDate, y2, m2, d2))
    {
        return 0;
    }
    return dayNumber(y2, m2, d2) - dayNumber(y1, m1, d1);
}

// Number of complete months from one date to another (never negative).
int fullMonthsBetween(const std::string& fromDate, const std::string& toDate)
{
    int y1 = 0, m1 = 0, d1 = 0, y2 = 0, m2 = 0, d2 = 0;
    if (!splitDate(fromDate, y1, m1, d1) || !splitDate(toDate, y2, m2, d2))
    {
        return 0;
    }
    int months = (y2 - y1) * 12 + (m2 - m1);
    if (d2 < d1)
    {
        --months;                  // the last month is not yet complete
    }
    return months < 0 ? 0 : months;
}

bool inputIsKeyboard()
{
    return INPUT_IS_TERMINAL();

// Reads one full line. When input is redirected from a file (automated
// testing) the typed value is echoed so the saved transcript shows it.
std::string readLine(const std::string& prompt)
{
    std::cout << prompt << std::flush;
    std::string line;
    if (!std::getline(std::cin, line))
    {
        throw InputClosedException();
    }
    if (!line.empty() && line[line.size() - 1] == '\r')
    {
        line.erase(line.size() - 1);      // remove Windows line ending
    }
    if (!inputIsKeyboard())
    {
        std::cout << line << "\n";
    }
    return line;
}

// Text must be printable and must not contain the '|' symbol, because '|'
// separates fields in the data files.
bool isSafeText(const std::string& value)
{
    for (std::size_t i = 0; i < value.size(); ++i)
    {
        unsigned char ch = static_cast<unsigned char>(value[i]);
        if (ch == static_cast<unsigned char>(Config::FIELD_DELIMITER) || ch < 32 || ch == 127)
        {
            return false;
        }
    }
    return true;
}

std::string readText(const std::string& prompt, std::size_t minLength, std::size_t maxLength)
{
    while (true)
    {
        std::string text = Utils::trim(readLine(prompt));
        if (text.size() < minLength)
        {
            Utils::printError(minLength <= 1 ? "This field cannot be empty."
                              : "Please type at least " + Utils::toString(static_cast<int>(minLength)) + " characters.");
        }
        else if (text.size() > maxLength)
        {
            Utils::printError("Please type at most " + Utils::toString(static_cast<int>(maxLength)) + " characters.");
        }
        else if (!isSafeText(text))
        {
            Utils::printError("The '|' symbol and control characters are not allowed.");
        }
        else
        {
            return text;
        }
    }
}

// Shows a limit in a friendly way for the error messages below.
static std::string describeLimit(double value, bool isMoney, bool wholeOnly)
{
    if (isMoney)
    {
        return Utils::formatUGX(value);
    }
    if (wholeOnly)
    {
        return Utils::toString(static_cast<int>(value));
    }
    return Utils::toString(value, 2);
}

// One core routine for every kind of number, so the checking rules are
// written once only (no repetition).
static double readNumberCore(const std::string& prompt, double minValue, double maxValue,
                             bool wholeOnly, bool isMoney, bool allowKeep, double currentValue)
{
    while (true)
    {
        std::string text = Utils::trim(readLine(prompt));
        if (isMoney)
        {
            text = Utils::removeChar(text, ',');   // accept "1,500,000"
        }
        if (text.empty())
        {
            if (allowKeep)
            {
                return currentValue;
            }
            Utils::printError(isMoney ? "Amount cannot be empty." : "A value is required.");
            continue;
        }

        double value = 0.0;
        if (!Utils::parseDouble(text, value))
        {
            Utils::printError(isMoney ? "Invalid amount. Type digits only, e.g. 50000."
                                      : "Invalid number. Type digits only.");
            continue;
        }
        if (value < 0)
        {
            Utils::printError(isMoney ? "Amount cannot be negative." : "Value cannot be negative.");
            continue;
        }
        if (wholeOnly && value != std::floor(value))
        {
            Utils::printError(isMoney ? "Enter whole shillings only (no decimals)." : "Please enter a whole number.");
            continue;
        }
        if (value < minValue)
        {
            Utils::printError("Value must be at least " + describeLimit(minValue, isMoney, wholeOnly) + ".");
            continue;
        }
        if (value > maxValue)
        {
            Utils::printError("Value must not be more than " + describeLimit(maxValue, isMoney, wholeOnly) + ".");
            continue;
        }
        return value;
    }
}

int readInt(const std::string& prompt, int minValue, int maxValue)
{
    return static_cast<int>(readNumberCore(prompt, minValue, maxValue, true, false, false, 0));
}

// Reads a password while showing '*' instead of the real characters.
std::string readPassword(const std::string& prompt)
{
    std::cout << prompt << std::flush;
    std::string password;

    if (!inputIsKeyboard())
    {
        // Input redirected from a file (automated tests): read normally.
        if (!std::getline(std::cin, password))
        {
            throw InputClosedException();
        }
        if (!password.empty() && password[password.size() - 1] == '\r')
        {
            password.erase(password.size() - 1);
        }
        std::cout << std::string(password.size(), '*') << "\n";
        return password;
    }

#ifdef _WIN32
    while (true)
    {
        int ch = _getch();
        if (ch == '\r' || ch == '\n')
        {
            break;
        }
        if (ch == 0 || ch == 224)
        {
            _getch();                  // ignore arrow and function keys
        }
        else if (ch == '\b')
        {
            if (!password.empty())
            {
                password.erase(password.size() - 1);
                std::cout << "\b \b" << std::flush;
            }
        }
        else if (ch >= 32 && ch < 127)
        {
            password += static_cast<char>(ch);
            std::cout << '*' << std::flush;
        }
    }
#else
    termios oldSettings;
    tcgetattr(STDIN_FILENO, &oldSettings);
    termios hiddenSettings = oldSettings;
    hiddenSettings.c_lflag &= ~static_cast<tcflag_t>(ECHO | ICANON);
    tcsetattr(STDIN_FILENO, TCSANOW, &hiddenSettings);

    while (true)
    {
        int ch = std::getchar();
        if (ch == EOF)
        {
            tcsetattr(STDIN_FILENO, TCSANOW, &oldSettings);
            throw InputClosedException();
        }
        if (ch == '\n' || ch == '\r')
        {
            break;
        }
        if (ch == 127 || ch == 8)      // backspace
        {
            if (!password.empty())
            {
                password.erase(password.size() - 1);
                std::cout << "\b \b" << std::flush;
            }
        }
        else if (ch >= 32 && ch < 127)
        {
            password += static_cast<char>(ch);
            std::cout << '*' << std::flush;
        }
    }
    tcsetattr(STDIN_FILENO, TCSANOW, &oldSettings);
#endif

    std::cout << "\n";
    return password;
}

// Full name: letters only (plus space ' - .), at least two names, 3-50 characters.
// Also tidies the name: extra spaces removed and each name starts with a capital.
bool validateName(std::string& value, std::string& error)
{
    std::string trimmed = Utils::trim(value);
    if (trimmed.empty())
    {
        error = "Name cannot be empty.";
        return false;
    }
    for (std::size_t i = 0; i < trimmed.size(); ++i)
    {
        char ch = trimmed[i];
        if (!std::isalpha(static_cast<unsigned char>(ch)) && ch != ' ' && ch != '\'' && ch != '-' && ch != '.')
        {
            error = "Name may contain letters, spaces, apostrophes (') and hyphens (-) only.";
            return false;
        }
    }

    std::istringstream words(trimmed);
    std::string word, cleaned;
    int wordCount = 0;
    while (words >> word)
    {
        word[0] = static_cast<char>(std::toupper(static_cast<unsigned char>(word[0])));
        cleaned += (wordCount > 0 ? " " : "") + word;
        ++wordCount;
    }
    if (cleaned.size() < 3 || cleaned.size() > 50)
    {
        error = "Name must be 3 to 50 characters long.";
        return false;
    }
    if (wordCount < 2)
    {
        error = "Please enter at least two names, e.g. Nakato Sarah.";
        return false;
    }
    value = cleaned;
    return true;
}

// Uganda National Identification Number: 14 letters/digits starting with
// CM (male) or CF (female), e.g. CM85012345ABCD. Stored in capital letters.
bool validateNIN(std::string& value, std::string& error)
{
    std::string nin = Utils::toUpper(Utils::removeChar(Utils::trim(value), ' '));
    bool ok = nin.size() == 14 && (nin.substr(0, 2) == "CM" || nin.substr(0, 2) == "CF");
    for (std::size_t i = 0; ok && i < nin.size(); ++i)
    {
        if (!std::isalnum(static_cast<unsigned char>(nin[i])))
        {
            ok = false;
        }
    }
    if (!ok)
    {
        error = "NIN must be 14 letters/digits starting with CM or CF, e.g. CM85012345ABCD.";
        return false;
    }
    value = nin;
    return true;
}

// Ugandan mobile number. Accepts 0772123456, +256772123456 or 256772123456
// and always stores the short form 0772123456.
bool validatePhone(std::string& value, std::string& error)
{
    std::string phone = Utils::removeChar(Utils::removeChar(Utils::trim(value), ' '), '-');
    if (phone.size() == 13 && phone.substr(0, 4) == "+256")
    {
        phone = "0" + phone.substr(4);
    }
    else if (phone.size() == 12 && phone.substr(0, 3) == "256")
    {
        phone = "0" + phone.substr(3);
    }
    if (phone.size() != 10 || !Utils::isAllDigits(phone) || phone.substr(0, 2) != "07")
    {
        error = "Phone must be a Ugandan mobile number, e.g. 0772123456 or +256772123456.";
        return false;
    }
    value = phone;
    return true;
}

bool validateAddress(std::string& value, std::string& error)
{
    std::string address = Utils::trim(value);
    if (address.size() < 3 || address.size() > 60)
    {
        error = "Address must be 3 to 60 characters long (e.g. Ndejje Central Ward).";
        return false;
    }
    if (!isSafeText(address))
    {
        error = "Address must not contain the '|' symbol.";
        return false;
    }
    value = address;
    return true;
}

// Staff usernames: 3-20 characters, letters/digits/underscore, start with a letter.
bool validateUsername(std::string& value, std::string& error)
{
    std::string name = Utils::toLower(Utils::trim(value));
    bool ok = name.size() >= 3 && name.size() <= 20 && std::isalpha(static_cast<unsigned char>(name[0]));
    for (std::size_t i = 0; ok && i < name.size(); ++i)
    {
        if (!std::isalnum(static_cast<unsigned char>(name[i])) && name[i] != '_')
        {
            ok = false;
        }
    }
    if (!ok)
    {
        error = "Username must be 3-20 letters, digits or _ and start with a letter.";
        return false;
    }
    value = name;
    return true;
}

bool Member::fromRecord(const std::string& line, Member& member)
{
    std::vector<std::string> field = Utils::split(line, Config::FIELD_DELIMITER);
    if (field.size() != 8 || field[0].empty() || !Utils::isValidDate(field[5]))
    {
        return false;
    }
    try
    {
        double shares  = Utils::numberOrThrow(field[6]);   // throws if the text is not a number
        double savings = Utils::numberOrThrow(field[7]);
        if (shares < 0 || savings < 0)
        {
            return false;
        }
        member = Member(field[0], field[1], field[2], field[3], field[4], field[5], shares, savings);
        return true;
    }
    catch (const std::invalid_argument&)
    {
        return false;               // a damaged number in the file: skip this line
    }
}

// Turns a password into a fixed 16-character code (FNV-1a hash, 1000 rounds,
// salted with the username). The real password is never saved to disk.
// NOTE: good enough for a classroom project; a bank-grade system would use
// a dedicated algorithm such as bcrypt or Argon2.
std::string StaffUser::hashPassword(const std::string& user, const std::string& password)
{
    const unsigned long long FNV_OFFSET = 1469598103934665603ULL;
    const unsigned long long FNV_PRIME  = 1099511628211ULL;
    const std::string data = user + ":" + password + ":NDEJJE-FARMERS-SACCO";

    unsigned long long hash = FNV_OFFSET;
    for (int round = 0; round < 1000; ++round)
    {
        for (std::size_t i = 0; i < data.size(); ++i)
        {
            hash ^= static_cast<unsigned char>(data[i]);
            hash *= FNV_PRIME;
        }
    }
    std::ostringstream out;
    out << std::hex << std::setw(16) << std::setfill('0') << hash;
    return out.str();
}

    return transactionId + d + memberId + d + typeToCode(type) + d + Utils::toString(amount, 2) + d +
           Utils::toString(balanceAfter, 2) + d + date + d + time + d + description + d + loanId + d +

bool Transaction::fromRecord(const std::string& line, Transaction& txn)
{
    std::vector<std::string> field = Utils::split(line, Config::FIELD_DELIMITER);
    TransactionType txnType;
    if (field.size() != 10 || field[0].empty() || !codeToType(field[2], txnType) ||
        !Utils::isValidDate(field[5]))
    {
        return false;
    }
    try
    {
        double txnAmount = Utils::numberOrThrow(field[3]);
        double balance   = Utils::numberOrThrow(field[4]);
        if (txnAmount < 0)
        {
            return false;
        }
        txn = Transaction(field[0], field[1], txnType, txnAmount, balance, field[5], field[6],
                          field[7], field[8], field[9]);
        return true;
    }
    catch (const std::invalid_argument&)   // damaged number in the file
    {
        return false;
    }
}

    {

    amountPaid += amount;
    if (getOutstandingBalance() <= MONEY_TOLERANCE)
    {
        status = LOAN_COMPLETED;
    }
    return true;
}

// Equal monthly installment (EMI) using the reducing-balance amortisation formula.
double Loan::calculateInstallment(double amount, double ratePercent, int months)
{
    if (amount <= 0 || months <= 0)
    {
        return 0.0;
    }
    double monthlyRate = ratePercent / 100.0 / 12.0;
    if (monthlyRate <= 0)
    {
        return Utils::roundMoney(amount / months);          // interest-free loan
    }
    double factor = std::pow(1.0 + monthlyRate, months);
    double installment = amount * monthlyRate * factor / (factor - 1.0);
    return Utils::roundMoney(installment);
}

// Builds the month-by-month repayment schedule. The last installment is
// adjusted so the balance ends at exactly zero after rounding.
std::vector<Installment> Loan::buildSchedule(double amount, double ratePercent, int months,
                                             const std::string& startDate)
{
    std::vector<Installment> schedule;
    if (amount <= 0 || months <= 0)
    {
        return schedule;
    }

    double monthlyRate = ratePercent / 100.0 / 12.0;
    double installment = calculateInstallment(amount, ratePercent, months);
    double balance = amount;

    for (int month = 1; month <= months; ++month)
    {
        Installment row;
        row.number = month;
        row.dueDate = startDate.empty() ? "" : Utils::addMonths(startDate, month);
        row.openingBalance = balance;
        row.interest = Utils::roundMoney(balance * monthlyRate);

        if (month == months)
        {
            row.principal = balance;                 // clear whatever is left
        }
        else
        {
            row.principal = installment - row.interest;
            if (row.principal > balance) row.principal = balance;
            if (row.principal < 0)       row.principal = 0;
        }
        row.payment = row.principal + row.interest;
        row.closingBalance = balance - row.principal;
        balance = row.closingBalance;
        schedule.push_back(row);
    }
    return schedule;
}

double Loan::totalPayableFor(double amount, double ratePercent, int months)
{
    std::vector<Installment> schedule = buildSchedule(amount, ratePercent, months, "");
    double total = 0.0;
    for (std::size_t i = 0; i < schedule.size(); ++i)
    {
        total += schedule[i].payment;
    }
    return total;
}

double Loan::getInstallment() const

double Loan::getOutstandingBalance() const
{
    if (status == LOAN_PENDING || status == LOAN_APPROVED || status == LOAN_REJECTED)
    {
        return 0.0;                         // nothing is owed until money is given out
    }
    double balance = getTotalPayable() - amountPaid;
    return balance > MONEY_TOLERANCE ? balance : 0.0;
}

// Total of all installments whose due date has arrived by the given date.
double Loan::getAmountDueBy(const std::string& date) const
{
    if (status != LOAN_ACTIVE && status != LOAN_COMPLETED)
    {
        return 0.0;
    }
    std::vector<Installment> schedule = getSchedule();
    double due = 0.0;
    for (std::size_t i = 0; i < schedule.size(); ++i)
    {
        if (!schedule[i].dueDate.empty() && schedule[i].dueDate <= date)
        {
            due += schedule[i].payment;
        }
    }
    return due;
}

// Money that should have been paid by 'date' but has not been paid.
double Loan::getArrears(const std::string& date) const
{
    if (status != LOAN_ACTIVE)
    {
        return 0.0;
    }
    double arrears = getAmountDueBy(date) - amountPaid;
    return arrears > MONEY_TOLERANCE ? Utils::roundMoney(arrears) : 0.0;
}

// Days since the oldest unpaid installment fell due.
int Loan::getDaysOverdue(const std::string& date) const
{
    if (getArrears(date) <= 0)
    {
        return 0;
    }
    std::vector<Installment> schedule = getSchedule();
    double cumulative = 0.0;
    for (std::size_t i = 0; i < schedule.size(); ++i)
    {
        cumulative += schedule[i].payment;
        if (cumulative > amountPaid + MONEY_TOLERANCE)
        {
            long days = Utils::daysBetween(schedule[i].dueDate, date);
            return days > 0 ? static_cast<int>(days) : 0;
        }
    }
    return 0;
}

// Due date of the oldest installment that is not yet fully paid.
std::string Loan::getNextDueDate() const
{
    if (status != LOAN_ACTIVE)
    {
        return "-";
    }
    std::vector<Installment> schedule = getSchedule();
    double cumulative = 0.0;
    for (std::size_t i = 0; i < schedule.size(); ++i)
    {
        cumulative += schedule[i].payment;
        if (cumulative > amountPaid + MONEY_TOLERANCE)
        {
            return schedule[i].dueDate;
        }
    }
    return "-";
}

// Standard microfinance classification of a loan by how late it is.
std::string Loan::arrearsCategory(int daysOverdue)
{
    if (daysOverdue <= 0)   return "Performing";
    if (daysOverdue <= 30)  return "Watch";
    if (daysOverdue <= 90)  return "Substandard";
    if (daysOverdue <= 180) return "Doubtful";
    return "Loss";
}

void Loan::printSchedule(std::ostream& out, const std::vector<Installment>& schedule,
                         double paidSoFar, bool showPaymentStatus)
{
    std::string today = Utils::today();
    out << "  " << std::left << std::setw(5) << "No." << std::setw(12) << "Due Date" << std::right
        << std::setw(13) << "Opening Bal" << std::setw(13) << "Installment" << std::setw(11) << "Interest"
        << std::setw(12) << "Principal" << std::setw(13) << "Closing Bal" << "  " << std::left
        << "Status\n";
    out << "  " << std::string(92, '-') << "\n";

    double cumulative = 0.0, totalPayment = 0.0, totalInterest = 0.0, totalPrincipal = 0.0;
    for (std::size_t i = 0; i < schedule.size(); ++i)
    {
        const Installment& row = schedule[i];
        double before = cumulative;
        cumulative += row.payment;

        std::string rowStatus = "-";
        if (showPaymentStatus)
        {
            if (paidSoFar + MONEY_TOLERANCE >= cumulative)        rowStatus = "PAID";
            else if (paidSoFar > before + MONEY_TOLERANCE)         rowStatus = "PART PAID";
            else if (!row.dueDate.empty() && row.dueDate <= today) rowStatus = "OVERDUE";
            else                                                   rowStatus = "UPCOMING";
        }

        out << "  " << std::left << std::setw(5) << row.number
            << std::setw(12) << (row.dueDate.empty() ? "Month " + Utils::toString(row.number) : row.dueDate)
            << std::right << std::setw(13) << Utils::formatMoney(row.openingBalance)
            << std::setw(13) << Utils::formatMoney(row.payment) << std::setw(11) << Utils::formatMoney(row.interest)
            << std::setw(12) << Utils::formatMoney(row.principal)
            << std::setw(13) << Utils::formatMoney(row.closingBalance) << "  " << std::left << rowStatus << "\n";

        totalPayment += row.payment;
        totalInterest += row.interest;
        totalPrincipal += row.principal;
    }
    out << "  " << std::string(92, '-') << "\n";
    out << "  " << std::left << std::setw(17) << "TOTALS" << std::right << std::setw(13) << ""
        << std::setw(13) << Utils::formatMoney(totalPayment) << std::setw(11) << Utils::formatMoney(totalInterest)
        << std::setw(12) << Utils::formatMoney(totalPrincipal) << "\n";
}

std::string Loan::statusToCode(LoanStatus loanStatus)
{
    switch (loanStatus)
    {
        case LOAN_PENDING:   return "PENDING";
        case LOAN_APPROVED:  return "APPROVED";
        case LOAN_ACTIVE:    return "ACTIVE";
        case LOAN_COMPLETED: return "COMPLETED";
        case LOAN_REJECTED:  return "REJECTED";
    }
    return "PENDING";
}

bool Loan::codeToStatus(const std::string& code, LoanStatus& loanStatus)
{
    if (code == "PENDING")        loanStatus = LOAN_PENDING;
    else if (code == "APPROVED")  loanStatus = LOAN_APPROVED;
    else if (code == "ACTIVE")    loanStatus = LOAN_ACTIVE;
    else if (code == "COMPLETED") loanStatus = LOAN_COMPLETED;
    else if (code == "REJECTED")  loanStatus = LOAN_REJECTED;
    else return false;
    return true;
}

std::string Loan::toRecord() const
{
    const char d = Config::FIELD_DELIMITER;
    return loanId + d + memberId + d + Utils::toString(principal, 2) + d +
           Utils::toString(annualInterestRate, 2) + d + Utils::toString(durationMonths) + d + purpose + d +
           statusToCode(status) + d + applicationDate + d + decisionDate + d + disbursementDate + d +
           Utils::toString(amountPaid, 2) + d + appliedBy + d + decidedBy + d + remarks;
}

bool Loan::fromRecord(const std::string& line, Loan& loan)
{
    std::vector<std::string> field = Utils::split(line, Config::FIELD_DELIMITER);
    LoanStatus loanStatus;
    if (field.size() != 14 || field[0].empty() || !codeToStatus(field[6], loanStatus) ||
        !Utils::isValidDate(field[7]))
    {
        return false;
    }
    // An active or completed loan must have a valid disbursement date.
    if ((loanStatus == LOAN_ACTIVE || loanStatus == LOAN_COMPLETED) && !Utils::isValidDate(field[9]))
    {
        return false;
    }
    try
    {
        loan = Loan(field[0], field[1], Utils::numberOrThrow(field[2]), Utils::numberOrThrow(field[3]),
                    Utils::wholeNumberOrThrow(field[4]), field[5], field[7], field[11]);
        loan.status           = loanStatus;
        loan.decisionDate     = field[8];
        loan.disbursementDate = field[9];
        loan.amountPaid       = Utils::numberOrThrow(field[10]);
        loan.decidedBy        = field[12];
        loan.remarks          = field[13];
        return loan.principal > 0 && loan.durationMonths > 0 && loan.amountPaid >= 0;
    }
    catch (const std::invalid_argument&)   // damaged number in the file
    {
        return false;
    }
}

// Applies one "key=value" line from the settings file. Bad values are ignored
// so a damaged file can never break the policy.
bool Settings::applyLine(const std::string& key, const std::string& value)
{
    double number = 0.0;
    if (!Utils::parseDouble(value, number))
    {
        return false;
    }
    int whole = static_cast<int>(number);

    if (key == "annual_interest_rate")      return setAnnualInterestRate(number);
    if (key == "loan_multiplier")           return setLoanMultiplier(number);
    if (key == "minimum_share_capital")     return setMinimumShareCapital(number);
    if (key == "minimum_savings")           return setMinimumSavings(number);
    if (key == "minimum_loan_amount")       return setLoanLimits(number, maximumLoanAmount > number ? maximumLoanAmount : number + 1);
    if (key == "maximum_loan_amount")       return setLoanLimits(minimumLoanAmount, number);
    if (key == "maximum_duration_months")   return setMaximumDurationMonths(whole);
    if (key == "minimum_membership_months") return setMinimumMembershipMonths(whole);
    if (key == "next_member_number" && whole >= 1)      { nextMemberNumber = whole; return true; }
    if (key == "next_loan_number" && whole >= 1)        { nextLoanNumber = whole; return true; }
    if (key == "next_transaction_number" && whole >= 1) { nextTransactionNumber = whole; return true; }
    return false;
}

/*
 * =============================================================================
 *  File    : FileManager.h
 *  Purpose : All reading and writing of data files happens here, so the rest
 *            of the program never deals with fstream details.
 *
 *  Safety features:
 *    - "Safe save": data is first written to a temporary file (*.tmp) and only
 *      then swapped in, so a power cut while saving cannot destroy the
 *      previous good copy.
 *    - Damaged lines are skipped and counted instead of crashing the program.
 *    - Lines starting with '#' are comments (column headings for humans).
 * =============================================================================
 */
#ifndef FILE_MANAGER_H
#define FILE_MANAGER_H

#include <string>
#include <vector>
#include "Settings.h"

class FileManager
{
public:
    static bool fileExists(const std::string& fileName);
    static bool readDataLines(const std::string& fileName, std::vector<std::string>& lines);
    static bool writeLinesSafely(const std::string& fileName, const std::vector<std::string>& lines);
    static bool appendLine(const std::string& fileName, const std::string& line);
    static std::vector<std::string> readLastLines(const std::string& fileName, std::size_t count);
    static bool copyFile(const std::string& source, const std::string& destination);

    static bool saveSettings(const std::string& fileName, const Settings& settings);
    static bool loadSettings(const std::string& fileName, Settings& settings);

    // Saves any list of records whose class provides toRecord().
    // (A template lets ONE function save members, loans, transactions and users.)
    template <typename T>
    static bool saveRecords(const std::string& fileName, const std::string& headerComment,
                            const std::vector<T>& records)
    {
        std::vector<std::string> lines;
        lines.push_back(headerComment);
        for (std::size_t i = 0; i < records.size(); ++i)
        {
            lines.push_back(records[i].toRecord());
        }
        return writeLinesSafely(fileName, lines);
    }

    // Loads any list of records whose class provides static fromRecord().
    // Returns the number loaded; damaged lines are counted in badLines.
    template <typename T>
    static int loadRecords(const std::string& fileName, std::vector<T>& records, int& badLines)
    {
        records.clear();
        badLines = 0;
        std::vector<std::string> lines;
        if (!readDataLines(fileName, lines))
        {
            return 0;                         // no file yet = empty list (first run)
        }
        for (std::size_t i = 0; i < lines.size(); ++i)
        {
            T item;
            if (T::fromRecord(lines[i], item))
            {
                records.push_back(item);
            }
            else
            {
                ++badLines;
            }
        }
        return static_cast<int>(records.size());
    }
};

bool FileManager::fileExists(const std::string& fileName)
{
    std::ifstream file(fileName.c_str());
    return file.good();
}

// Reads all useful lines (skips blank lines and '#' comments).
// If the main file is missing but a .tmp copy survived an interrupted save,
// the .tmp copy is used so no data is lost.
bool FileManager::readDataLines(const std::string& fileName, std::vector<std::string>& lines)
{
    lines.clear();
    std::string source = fileName;
    if (!fileExists(source) && fileExists(fileName + ".tmp"))
    {
        source = fileName + ".tmp";
    }

    std::ifstream file(source.c_str());
    if (!file)
    {
        return false;
    }
    std::string line;
    while (std::getline(file, line))
    {
        if (!line.empty() && line[line.size() - 1] == '\r')
        {
            line.erase(line.size() - 1);        // file edited on Windows
        }
        if (Utils::trim(line).empty() || line[0] == '#')
        {
            continue;
        }
        lines.push_back(line);
    }
    return true;
}

// Writes to "<name>.tmp" first, then replaces the real file.
bool FileManager::writeLinesSafely(const std::string& fileName, const std::vector<std::string>& lines)
{
    const std::string tempName = fileName + ".tmp";
    {
        std::ofstream out(tempName.c_str(), std::ios::out | std::ios::trunc);
        if (!out)
        {
            return false;
        }
        for (std::size_t i = 0; i < lines.size(); ++i)
        {
            out << lines[i] << "\n";
        }
        out.flush();
        if (!out)
        {
            return false;                       // e.g. disk full
        }
    }   // the file is closed here

    std::remove(fileName.c_str());              // Windows cannot rename over an existing file
    return std::rename(tempName.c_str(), fileName.c_str()) == 0;
}

bool FileManager::appendLine(const std::string& fileName, const std::string& line)
{
    std::ofstream out(fileName.c_str(), std::ios::out | std::ios::app);
    if (!out)
    {
        return false;
    }
    out << line << "\n";
    return out.good();
}

// Returns only the last 'count' lines (used to show the latest audit entries).
std::vector<std::string> FileManager::readLastLines(const std::string& fileName, std::size_t count)
{
    std::deque<std::string> window;
    std::ifstream file(fileName.c_str());
    std::string line;
    while (std::getline(file, line))
    {
        if (Utils::trim(line).empty())
        {
            continue;
        }
        window.push_back(line);
        if (window.size() > count)
        {
            window.pop_front();
        }
    }
    return std::vector<std::string>(window.begin(), window.end());
}

bool FileManager::copyFile(const std::string& source, const std::string& destination)
{
    std::ifstream in(source.c_str(), std::ios::binary);
    if (!in)
    {
        return false;
    }
    std::ofstream out(destination.c_str(), std::ios::binary | std::ios::trunc);
    if (!out)
    {
        return false;
    }
    out << in.rdbuf();
    return out.good();
}

bool FileManager::saveSettings(const std::string& fileName, const Settings& settings)
{
    return writeLinesSafely(fileName, settings.toLines());
}

bool FileManager::loadSettings(const std::string& fileName, Settings& settings)
{
    std::vector<std::string> lines;
    if (!readDataLines(fileName, lines))
    {
        return false;                           // first run: defaults stay in place
    }
    for (std::size_t i = 0; i < lines.size(); ++i)
    {
        std::size_t equals = lines[i].find('=');
        if (equals != std::string::npos)
        {
            settings.applyLine(Utils::trim(lines[i].substr(0, equals)), Utils::trim(lines[i].substr(equals + 1)));
        }
    }
    return true;
}

    // Column headings written at the top of each data file (for human readers).
    const std::string MEMBERS_HEADER =
        "# MemberID|FullName|NIN|Phone|Address|DateJoined|ShareCapital|SavingsBalance";
    const std::string LOANS_HEADER =
        "# LoanID|MemberID|Principal|InterestRate|Months|Purpose|Status|AppliedOn|DecisionDate|"
        "DisbursedOn|AmountPaid|AppliedBy|DecidedBy|Remarks";
    const std::string TRANSACTIONS_HEADER =
        "# TxnID|MemberID|Type|Amount|BalanceAfter|Date|Time|Description|LoanID|RecordedBy";
    const std::string USERS_HEADER = "# Username|PasswordHash|Role|FullName|Phone|Active";

    // "NFS0012" with prefix length 3 -> 12. Returns 0 if there is no number.
    int idNumber(const std::string& id, std::size_t prefixLength)
    {
        int number = 0;
        if (id.size() > prefixLength && Utils::parseInt(id.substr(prefixLength), number))
        {
            return number;
        }
        return 0;
    }

    // Sort rule for the ledger: earlier date/time first.
    bool happensEarlier(const Transaction& first, const Transaction& second)
    {
        return first.getDate() + first.getTime() < second.getDate() + second.getTime();
    }

    // Keeps a copy of a data file that contained damaged lines, before it is rewritten.
    void keepDamagedCopy(const std::string& fileName, int badLines)
    {
        if (badLines > 0)
        {
            std::string copyName = fileName + ".damaged_" + Utils::removeChar(Utils::today(), '-');
            FileManager::copyFile(fileName, copyName);
            Utils::printWarning(Utils::toString(badLines) + " damaged line(s) skipped in " + fileName +
                                ". Original kept as " + copyName);
        }
    }
}

void SaccoSystem::loadAllData()
{
    int badMembers = 0, badLoans = 0, badTransactions = 0, badUsers = 0;

    FileManager::loadSettings(Config::SETTINGS_FILE, settings);   // missing file = default policy
    FileManager::loadRecords(Config::MEMBERS_FILE, members, badMembers);
    FileManager::loadRecords(Config::LOANS_FILE, loans, badLoans);
    FileManager::loadRecords(Config::TRANSACTIONS_FILE, transactions, badTransactions);
    FileManager::loadRecords(Config::USERS_FILE, users, badUsers);

    keepDamagedCopy(Config::MEMBERS_FILE, badMembers);
    keepDamagedCopy(Config::LOANS_FILE, badLoans);
    keepDamagedCopy(Config::TRANSACTIONS_FILE, badTransactions);
    keepDamagedCopy(Config::USERS_FILE, badUsers);

    // Make sure newly generated IDs can never repeat an ID already in the files.
    int highestMember = 0, highestLoan = 0, highestTransaction = 0;
    for (std::size_t i = 0; i < members.size(); ++i)
        highestMember = std::max(highestMember, idNumber(members[i].getMemberId(), 3));
    for (std::size_t i = 0; i < loans.size(); ++i)
        highestLoan = std::max(highestLoan, idNumber(loans[i].getLoanId(), 2));
    for (std::size_t i = 0; i < transactions.size(); ++i)
        highestTransaction = std::max(highestTransaction, idNumber(transactions[i].getTransactionId(), 2));
    settings.ensureCountersAbove(highestMember, highestLoan, highestTransaction);

    if (users.empty())
    {
        createDefaultUsers();
    }
    FileManager::saveSettings(Config::SETTINGS_FILE, settings);
}

// Saves every data file. Called after EVERY change so nothing is lost if the
// computer switches off. Each file is saved even if an earlier one fails.
bool SaccoSystem::saveAll()
{
    bool membersSaved      = FileManager::saveRecords(Config::MEMBERS_FILE, MEMBERS_HEADER, members);
    bool loansSaved        = FileManager::saveRecords(Config::LOANS_FILE, LOANS_HEADER, loans);
    bool transactionsSaved = FileManager::saveRecords(Config::TRANSACTIONS_FILE, TRANSACTIONS_HEADER, transactions);
    bool usersSaved        = FileManager::saveRecords(Config::USERS_FILE, USERS_HEADER, users);
    bool settingsSaved     = FileManager::saveSettings(Config::SETTINGS_FILE, settings);

    bool allSaved = membersSaved && loansSaved && transactionsSaved && usersSaved && settingsSaved;
    if (!allSaved)
    {
        Utils::printError("Some data could not be saved. Check that the program folder is not "
                          "read-only and the disk is not full.");
    }
    return allSaved;
}

void SaccoSystem::run()
{
    bool finished = false;
    while (!finished)
    {
        try
        {
            welcomeMenu();
            finished = true;                       // user chose Exit
        }
        catch (const InputClosedException&)
        {
            std::cout << "\n";
            Utils::printInfo("Input closed. All data has been saved. Goodbye.");
            finished = true;
        }
        catch (const std::exception& error)
        {
            // Last safety net: report the problem and return to the welcome screen.
            Utils::printError(std::string("Unexpected problem: ") + error.what() + ". Returning to the welcome screen.");
            currentUserIndex = -1;
        }
    }
    saveAll();
}

bool SaccoSystem::staffLogin()
{
    showBanner("Staff Login");
    for (int attempt = 1; attempt <= Config::MAX_LOGIN_ATTEMPTS; ++attempt)
    {
        std::string username = Utils::toLower(Utils::trim(InputHelper::readLine("  Username: ")));
        std::string password = InputHelper::readPassword("  Password: ");
        int index = findUserIndex(username);

        if (index >= 0 && users[index].checkPassword(password))
        {
            if (!users[index].isActive())
            {
                Utils::printError("This account has been deactivated. Please contact the Administrator.");
                InputHelper::pause();
                return false;
            }
            currentUserIndex = index;
            firstRun = false;
            logAudit("Logged in");
            Utils::printSuccess("Welcome, " + users[index].getFullName() + "!");
            if (users[index].checkPassword(users[index].getUsername() + "123"))
            {
                Utils::printWarning("You are using the default password. Please change it (Main Menu > 7).");
            }
            InputHelper::pause();
            return true;
        }

        int attemptsLeft = Config::MAX_LOGIN_ATTEMPTS - attempt;
        Utils::printError("Wrong username or password." +
                          (attemptsLeft > 0 ? " Attempts left: " + Utils::toString(attemptsLeft) : std::string("")));
    }
    Utils::printError("Too many failed attempts. Returning to the welcome screen.");
    FileManager::appendLine(Config::AUDIT_LOG_FILE, Utils::today() + " " + Utils::currentTime() +
                            " | " + Utils::padRight("unknown", 12) + " | Login blocked after 3 failed attempts");
    InputHelper::pause();
    return false;
}

        switch (choice)
        {
            case 1:
                Utils::printTitle("Current loan policy");
                settings.display(std::cout);
                break;
            case 2: updateLoanPolicy();       break;
            case 3: addStaffUser();           break;
            case 4: listStaffUsers();         break;
            case 5: resetStaffPassword();     break;
            case 6: toggleStaffStatus();      break;
            case 7: backupData();             break;
            case 8: viewAuditLog();           break;
            case 9: loadDemonstrationData();  break;
        }
        if (choice != 0)
        {
            InputHelper::pause();
        }
    } while (choice != 0);
}

// Asks for a new password twice. Returns "" if the user fails 3 times.
std::string SaccoSystem::readNewPassword(const std::string& username)
{
    for (int attempt = 1; attempt <= 3; ++attempt)
    {
        std::string password = InputHelper::readPassword("  New password (6-30 characters): ");
        if (password.size() < 6 || password.size() > 30)
        {
            Utils::printError("Password must be 6 to 30 characters long.");
            continue;
        }
        if (Utils::equalsIgnoreCase(password, username))
        {
            Utils::printError("Password must not be the same as the username.");
            continue;
        }
        std::string confirm = InputHelper::readPassword("  Confirm new password: ");
        if (confirm != password)
        {
            Utils::printError("The two passwords do not match.");
            continue;
        }
        return password;
    }
    Utils::printError("Password not set after 3 attempts.");
    return "";
}

void SaccoSystem::updateLoanPolicy()
{
    Utils::printTitle("Update loan policy");
    settings.display(std::cout);
    std::cout << "\n";
    Utils::printInfo("Press Enter to keep the current value shown in [brackets].");

    double rate     = InputHelper::readDecimalOrKeep("  Interest rate (% per year)", settings.getAnnualInterestRate(), 0, 60);
    double multiple = InputHelper::readDecimalOrKeep("  Loan multiplier (x savings + shares)", settings.getLoanMultiplier(), 1, 10);
    double minShare = InputHelper::readAmountOrKeep("  Minimum share capital to borrow (UGX)", settings.getMinimumShareCapital(), 0, 10000000);
    double minSave  = InputHelper::readAmountOrKeep("  Minimum savings to borrow (UGX)", settings.getMinimumSavings(), 0, 10000000);
    double minLoan  = InputHelper::readAmountOrKeep("  Smallest loan allowed (UGX)", settings.getMinimumLoanAmount(), 10000, 500000000);
    double maxLoan  = InputHelper::readAmountOrKeep("  Largest loan allowed (UGX)", settings.getMaximumLoanAmount(), 10000, 500000000);
    int maxMonths   = InputHelper::readIntOrKeep("  Longest repayment period (months)", settings.getMaximumDurationMonths(), 1, 60);
    int minMember   = InputHelper::readIntOrKeep("  Minimum membership period (months)", settings.getMinimumMembershipMonths(), 0, 24);

    if (minLoan >= maxLoan)
    {
        Utils::printError("The smallest loan must be less than the largest loan. No changes were saved.");
        return;
    }
    if (!InputHelper::readYesNo("  Save these policy changes?"))
    {
        Utils::printInfo("No changes were made.");
        return;
    }
    double oldRate = settings.getAnnualInterestRate();
    settings.setAnnualInterestRate(rate);
    settings.setLoanMultiplier(multiple);
    settings.setMinimumShareCapital(minShare);
    settings.setMinimumSavings(minSave);
    settings.setLoanLimits(minLoan, maxLoan);
    settings.setMaximumDurationMonths(maxMonths);
    settings.setMinimumMembershipMonths(minMember);
    saveAll();
    logAudit("Updated loan policy (interest " + Utils::toString(oldRate, 2) + "% -> " + Utils::toString(rate, 2) + "%)");
    Utils::printSuccess("Loan policy updated. Existing loans keep the interest rate they were given.");
}

void SaccoSystem::toggleStaffStatus()
{
    Utils::printTitle("Activate / deactivate a staff user");
    std::string username = Utils::toLower(Utils::trim(InputHelper::readLine("  Username: ")));
    int index = findUserIndex(username);
    if (index < 0)
    {
        Utils::printError("No staff user called '" + username + "'.");
        return;
    }
    if (index == currentUserIndex)
    {
        Utils::printError("You cannot deactivate your own account.");
        return;
    }
    StaffUser& user = users[index];
    if (user.isActive() && user.isAdministrator())
    {
        int activeAdmins = 0;
        for (std::size_t i = 0; i < users.size(); ++i)
        {
            if (users[i].isActive() && users[i].isAdministrator()) ++activeAdmins;
        }
        if (activeAdmins <= 1)
        {
            Utils::printError("At least one active Administrator must remain.");
            return;
        }
    }
    std::string action = user.isActive() ? "Deactivate" : "Activate";
    if (!InputHelper::readYesNo("  " + action + " '" + username + "'?"))
    {
        Utils::printInfo("No changes were made.");
        return;
    }
    user.setActive(!user.isActive());
    saveAll();
    logAudit(action + "d staff user " + username);
    Utils::printSuccess("User '" + username + "' is now " + (user.isActive() ? "active." : "deactivated."));
}

// Copies every data file to a time-stamped backup copy.
void SaccoSystem::backupData()
{
    Utils::printTitle("Back up data files");
    saveAll();
    std::string stamp = Utils::removeChar(Utils::today(), '-') + "_" + Utils::removeChar(Utils::currentTime(), ':');
    const std::string files[] = { Config::MEMBERS_FILE, Config::LOANS_FILE, Config::TRANSACTIONS_FILE,
                                  Config::USERS_FILE, Config::SETTINGS_FILE, Config::AUDIT_LOG_FILE };
    const int fileCount = sizeof(files) / sizeof(files[0]);
    int copied = 0;
    for (int i = 0; i < fileCount; ++i)
    {
        if (!FileManager::fileExists(files[i]))
        {
            continue;
        }
        std::string backupName = "backup_" + stamp + "_" + files[i];
        if (FileManager::copyFile(files[i], backupName))
        {
            std::cout << "  Copied " << files[i] << "  ->  " << backupName << "\n";
            ++copied;
        }
        else
        {
            Utils::printError("Could not copy " + files[i]);
        }
    }
    logAudit("Backed up " + Utils::toString(copied) + " data file(s)");
    Utils::printSuccess(Utils::toString(copied) + " file(s) backed up. Store copies on a flash disk as well.");
}

// Fills an EMPTY system with realistic sample records so every feature
// (including arrears and completed loans) can be demonstrated at once.
void SaccoSystem::loadDemonstrationData()
{
    Utils::printTitle("Load demonstration data");
    if (!members.empty() || !loans.empty() || !transactions.empty())
    {
        Utils::printError("Demonstration data can only be loaded into an empty system.");
        return;
    }
    Utils::printInfo("This adds 6 sample members, their savings and shares, and 5 loans at different stages.");
    if (!InputHelper::readYesNo("  Load demonstration data now?"))
    {
        Utils::printInfo("Cancelled.");
        return;
    }

    const std::string today = Utils::today();

    struct DemoMember
    {
        const char* name;
        const char* nin;
        const char* phone;
        const char* address;
        int monthsAgo;
        double shares;
        double savings;
    };
    const DemoMember demo[] =
    {
        { "Ssemakula John Bosco", "CM78001234ABCD", "0772123456", "Ndejje Central Ward",   24, 200000, 600000 },
        { "Nakato Florence",      "CF85023456BCDE", "0701234567", "Ndejje North Ward",     18, 150000, 400000 },
        { "Kizza Richard",        "CM90034567CDEF", "0756345678", "Ndejje South Ward",     30, 100000, 250000 },
        { "Namubiru Agnes",       "CF88045678DEFG", "0782456789", "Ndejje East Ward",      12, 120000, 300000 },
        { "Mukasa Peter",         "CM92056789EFGH", "0703567890", "Ndejje West Ward",       8, 100000, 180000 },
        { "Nansubuga Ruth",       "CF95067890FGHI", "0774678901", "Ndejje Trading Centre",  1,  40000,  30000 }
    };
    const int demoCount = sizeof(demo) / sizeof(demo[0]);

    for (int i = 0; i < demoCount; ++i)
    {
        std::string id = settings.generateMemberId();
        std::string joined = Utils::addMonths(today, -demo[i].monthsAgo);
        Member member(id, demo[i].name, demo[i].nin, demo[i].phone, demo[i].address, joined, 0, 0);

        member.addShareCapital(demo[i].shares);
        addTransactionOn(joined, "09:00:00", id, TXN_SHARE_PURCHASE, demo[i].shares, member.getShareCapital(),
                         "Shares at registration", "");
        double firstDeposit = demo[i].savings / 2;
        member.depositSavings(firstDeposit);
        addTransactionOn(joined, "09:15:00", id, TXN_SAVINGS_DEPOSIT, firstDeposit, member.getSavingsBalance(),
                         "Opening savings deposit", "");
        member.depositSavings(demo[i].savings - firstDeposit);
        addTransactionOn(Utils::addMonths(joined, 1), "10:00:00", id, TXN_SAVINGS_DEPOSIT,
                         demo[i].savings - firstDeposit, member.getSavingsBalance(), "Cash deposit", "");
        members.push_back(member);
    }

    // Loans: {member index, amount, months, purpose, months since payout (-1 = not paid out),
    //         installments repaid, final stage}
    struct DemoLoan
    {
        int memberIndex;
        double amount;
        int months;
        const char* purpose;
        int disbursedMonthsAgo;
        int installmentsPaid;
        bool approveOnly;
    };
    const DemoLoan demoLoans[] =
    {
        { 0, 2000000, 12, "Livestock or poultry - two dairy heifers",        5, 5,  false },  // active, up to date
        { 1, 1500000,  6, "Crop farming inputs - maize seed and fertiliser", 4, 1,  false },  // active, in arrears
        { 2,  800000,  6, "Farm tools or equipment - irrigation pump",       9, 6,  false },  // completed
        { 3, 1000000, 10, "Produce trade or storage - coffee drying shed",  -1, 0,  false },  // pending
        { 4,  600000,  6, "School fees - secondary school fees",            -1, 0,  true  }   // approved
    };
    const int loanCount = sizeof(demoLoans) / sizeof(demoLoans[0]);

    for (int i = 0; i < loanCount; ++i)
    {
        const DemoLoan& d = demoLoans[i];
        const Member& member = members[d.memberIndex];
        std::string payoutDate = d.disbursedMonthsAgo >= 0 ? Utils::addMonths(today, -d.disbursedMonthsAgo) : today;
        Loan loan(settings.generateLoanId(), member.getMemberId(), d.amount, settings.getAnnualInterestRate(),
                  d.months, d.purpose, payoutDate, "officer");

        if (d.approveOnly || d.disbursedMonthsAgo >= 0)
        {
            loan.approve(payoutDate, "admin", "Meets all eligibility rules");
        }
        if (d.disbursedMonthsAgo >= 0)
        {
            loan.disburse(payoutDate);
            addTransactionOn(payoutDate, "11:00:00", member.getMemberId(), TXN_LOAN_DISBURSEMENT, d.amount,
                             loan.getTotalPayable(), "Loan paid out", loan.getLoanId());
            std::vector<Installment> schedule = loan.getSchedule();
            for (int k = 0; k < d.installmentsPaid && k < static_cast<int>(schedule.size()); ++k)
            {
                loan.recordRepayment(schedule[k].payment);
                addTransactionOn(schedule[k].dueDate, "14:00:00", member.getMemberId(), TXN_LOAN_REPAYMENT,
                                 schedule[k].payment, loan.getOutstandingBalance(),
                                 "Installment " + Utils::toString(k + 1), loan.getLoanId());
            }
        }
        loans.push_back(loan);
    }

    // Put the ledger in date order and number the transactions in that order.
    std::stable_sort(transactions.begin(), transactions.end(), happensEarlier);
    settings.resetTransactionCounter(1);
    for (std::size_t i = 0; i < transactions.size(); ++i)
    {
        transactions[i].setTransactionId(settings.generateTransactionId());
    }

    saveAll();
    logAudit("Loaded demonstration data");
    Utils::printSuccess("Loaded " + Utils::toString(demoCount) + " members, " + Utils::toString(loanCount) +
                        " loans and " + Utils::toString(static_cast<int>(transactions.size())) + " transactions.");
}

// Asks for a Member ID. Typing only the number (e.g. 1) is accepted as NFS0001.
int SaccoSystem::promptMemberIndex()
{
    std::string input = Utils::toUpper(Utils::trim(InputHelper::readLine("  Enter Member ID (e.g. NFS0001): ")));
    if (input.empty())
    {
        Utils::printError("Member ID cannot be empty.");
        return -1;
    }
    int number = 0;
    if (Utils::isAllDigits(input) && Utils::parseInt(input, number))
    {
        input = "NFS" + Utils::zeroPad(number, 4);
    }
    int index = findMemberIndex(input);
    if (index < 0)
    {
        Utils::printError("No member found with ID '" + input + "'. Please check the ID and try again.");
    }
    return index;
}

// Asks for a Loan ID, or a Member ID (then that member's unfinished loan,
// or their most recent loan, is used).
int SaccoSystem::promptLoanIndex()
{
    std::string input = Utils::toUpper(Utils::trim(
        InputHelper::readLine("  Enter Loan ID (e.g. LN0001) or Member ID (e.g. NFS0001): ")));
    if (input.empty())
    {
        Utils::printError("Loan ID cannot be empty.");
        return -1;
    }
    int number = 0;
    if (Utils::isAllDigits(input) && Utils::parseInt(input, number))
    {
        input = "LN" + Utils::zeroPad(number, 4);
    }
    if (input.compare(0, 3, "NFS") == 0)
    {
        if (findMemberIndex(input) < 0)
        {
            Utils::printError("No member found with ID '" + input + "'.");
            return -1;
        }
        int open = findOpenLoanIndex(input, "");
        if (open >= 0)
        {
            return open;
        }
        for (int i = static_cast<int>(loans.size()) - 1; i >= 0; --i)
        {
            if (loans[i].getMemberId() == input)
            {
                return i;
            }
        }
        Utils::printError("Member " + input + " has no loans.");
        return -1;
    }
    int index = findLoanIndex(input);
    if (index < 0)
    {
        Utils::printError("No loan found with ID '" + input + "'. Please check the ID and try again.");
    }
    return index;
}

{
    transactions.push_back(Transaction(settings.generateTransactionId(), memberId, type, amount, balanceAfter,
                                       date, time, description, loanId, currentUsername()));
}

void SaccoSystem::registerMember()
{
    Utils::printTitle("Register new member");
    Utils::printInfo("Opening balances may be entered for members moving from the paper registers.");

    std::string name = InputHelper::readValidated("  Full name (e.g. Nakato Sarah)       : ", InputHelper::validateName);
    std::string nin  = InputHelper::readValidated("  National ID Number (NIN)            : ", InputHelper::validateNIN);
    int existing = findMemberIndexByNIN(nin);
    if (existing >= 0)
    {
        Utils::printError("This NIN is already registered to " + members[existing].getMemberId() + " (" +
                          members[existing].getFullName() + "). Registration cancelled.");
        return;
    }
    std::string phone   = InputHelper::readValidated("  Phone number (e.g. 0772123456)      : ", InputHelper::validatePhone);
    std::string address = InputHelper::readValidated("  Address / village                   : ", InputHelper::validateAddress);
    std::string today   = Utils::today();
    std::string joined  = InputHelper::readDate("  Date joined (YYYY-MM-DD)", today, Config::EARLIEST_DATE, today);

    std::cout << "  Each share costs " << Utils::formatUGX(Config::SHARE_PRICE) << ". At least 1 share must be bought.\n";
    int shares = InputHelper::readInt("  Number of shares bought             : ", 1, 2500);
    double openingSavings = InputHelper::readAmount("  Opening savings deposit (UGX, 0 = none): ", 0, Config::MAX_TRANSACTION);
    double shareAmount = shares * Config::SHARE_PRICE;

    // Show a summary and ask for confirmation before anything is saved.
    Utils::printTitle("Please confirm");
    std::cout << "  Name          : " << name << "\n"
              << "  NIN           : " << nin << "\n"
              << "  Phone         : " << phone << "\n"
              << "  Address       : " << address << "\n"
              << "  Date joined   : " << joined << "\n"
              << "  Share capital : " << Utils::formatUGX(shareAmount) << " (" << shares << " shares)\n"
              << "  Savings       : " << Utils::formatUGX(openingSavings) << "\n";
    if (!InputHelper::readYesNo("  Save this member?"))
    {
        Utils::printInfo("Registration cancelled. Nothing was saved.");
        return;
    }

    std::string id = settings.generateMemberId();
    Member member(id, name, nin, phone, address, joined, 0, 0);
    member.addShareCapital(shareAmount);
    recordTransaction(id, TXN_SHARE_PURCHASE, shareAmount, member.getShareCapital(),
                      "Shares at registration (" + Utils::toString(shares) + ")", "");
    if (openingSavings > 0)
    {
        member.depositSavings(openingSavings);
        recordTransaction(id, TXN_SAVINGS_DEPOSIT, openingSavings, member.getSavingsBalance(),
                          "Opening savings deposit", "");
    }
    members.push_back(member);
    saveAll();
    logAudit("Registered member " + id + " (" + name + ")");
    Utils::printSuccess("Member registered successfully. Member ID: " + id);
}

// Profile + loan position + how much the member could borrow.

        if (loans[i].getMemberId() == member.getMemberId()) ++loanCount;

    std::cout << "  Loans on record  : " << loanCount << "\n";

    int open = findOpenLoanIndex(member.getMemberId(), "");
    if (open >= 0)
    {
        const Loan& loan = loans[open];
        std::cout << "  Current loan     : " << loan.getLoanId() << " - " << loan.getStatusName()

void SaccoSystem::deleteMember()
{
    Utils::printTitle("Delete member");
    int index = promptMemberIndex();
    if (index < 0)
    {
        return;
    }
    std::string id = members[index].getMemberId();
    std::string name = members[index].getFullName();
    std::cout << "\n";
    members[index].displayProfile(std::cout);

    int open = findOpenLoanIndex(id, "");
    if (open >= 0)
    {
        Utils::printError("Cannot delete: the member has an unfinished loan " + loans[open].getLoanId() + " (" +
                          loans[open].getStatusName() + "). Complete or cancel it first.");
        return;
    }
    double refund = members[index].getTotalFunds();
    if (refund > 0)
    {
        Utils::printWarning("Savings and share capital of " + Utils::formatUGX(refund) +
                            " will be refunded to the member (account closure).");
    }
    Utils::printInfo("Past transactions and finished loans are kept for the auditors.");

    {
        members[index].closeAccount();
        recordTransaction(id, TXN_ACCOUNT_CLOSURE, refund, 0, "Refund on account closure", "");
    }
    members.erase(members.begin() + index);
    saveAll();
    logAudit("Deleted member " + id + " (" + name + "), refunded " + Utils::formatUGX(refund));
    Utils::printSuccess("Member " + id + " (" + name + ") has been deleted.");

    // Adds one rule to the checklist; any failed rule makes the member ineligible.
    void addRule(EligibilityResult& result, const std::string& description, bool passed, const std::string& detail)
    {
        EligibilityRule rule;
        rule.description = description;
        rule.passed = passed;
        rule.detail = detail;
        result.rules.push_back(rule);
        if (!passed)
        {
            result.eligible = false;
        }
    }
}

// Largest loan the member could get: multiplier x (savings + shares), capped by
// the policy maximum and rounded down to whole thousands.
double SaccoSystem::maximumLoanFor(const Member& member) const
{
    double limit = settings.getLoanMultiplier() * member.getTotalFunds();
    if (limit > settings.getMaximumLoanAmount())
    {
        limit = settings.getMaximumLoanAmount();
    }
    return std::floor(limit / 1000.0) * 1000.0;
}

EligibilityResult SaccoSystem::checkEligibility(const Member& member, double amount, int months,
                                                const std::string& excludeLoanId) const
{
    EligibilityResult result;
    result.eligible = true;
    result.maximumLoan = maximumLoanFor(member);

    int monthsAsMember = Utils::fullMonthsBetween(member.getDateJoined(), Utils::today());
    addRule(result, "Member for at least " + Utils::toString(settings.getMinimumMembershipMonths()) + " month(s)",
            monthsAsMember >= settings.getMinimumMembershipMonths(),
            "Member for " + Utils::toString(monthsAsMember) + " month(s)");

    addRule(result, "Share capital at least " + Utils::formatUGX(settings.getMinimumShareCapital()),
            member.getShareCapital() >= settings.getMinimumShareCapital(),
            "Has " + Utils::formatUGX(member.getShareCapital()));

    addRule(result, "Savings at least " + Utils::formatUGX(settings.getMinimumSavings()),
            member.getSavingsBalance() >= settings.getMinimumSavings(),
            "Has " + Utils::formatUGX(member.getSavingsBalance()));

    int open = findOpenLoanIndex(member.getMemberId(), excludeLoanId);
    addRule(result, "No other unfinished loan", open < 0,
            open < 0 ? "None" : loans[open].getLoanId() + " is " + loans[open].getStatusName());

    addRule(result, "Amount between " + Utils::formatMoney(settings.getMinimumLoanAmount()) + " and " +
                    Utils::formatMoney(settings.getMaximumLoanAmount()),
            amount >= settings.getMinimumLoanAmount() && amount <= settings.getMaximumLoanAmount(),
            "Asked for " + Utils::formatUGX(amount));

    double fundsLimit = settings.getLoanMultiplier() * member.getTotalFunds();
    addRule(result, "Amount within " + Utils::toString(settings.getLoanMultiplier(), 1) + " x (savings + shares)",
            amount <= fundsLimit, "Limit " + Utils::formatUGX(fundsLimit));

    addRule(result, "Period of 1 to " + Utils::toString(settings.getMaximumDurationMonths()) + " months",
            months >= 1 && months <= settings.getMaximumDurationMonths(),
            "Asked for " + Utils::toString(months) + " month(s)");

    return result;
}

void SaccoSystem::displayEligibility(const EligibilityResult& result) const
{
    Utils::printTitle("Eligibility checklist");
    std::cout << "  " << std::left << std::setw(4) << "No" << std::setw(44) << "Rule" << std::setw(8) << "Result"
              << "Details\n";
    std::cout << "  " << std::string(90, '-') << "\n";
    int failed = 0;
    for (std::size_t i = 0; i < result.rules.size(); ++i)
    {
        const EligibilityRule& rule = result.rules[i];
        std::cout << "  " << std::left << std::setw(4) << (i + 1) << std::setw(44) << Utils::fitWidth(rule.description, 43)
                  << std::setw(8) << (rule.passed ? "PASS" : "FAIL") << rule.detail << "\n";
        if (!rule.passed)
        {
            ++failed;
        }
    }
    std::cout << "  " << std::string(90, '-') << "\n";
    if (result.eligible)
    {
        std::cout << "  RESULT: ELIGIBLE - all " << result.rules.size() << " rules passed.\n";
    }
    else
    {
        std::cout << "  RESULT: NOT ELIGIBLE - " << failed << " rule(s) failed (marked FAIL).\n";
    }
    std::cout << "  Most this member can borrow now: " << Utils::formatUGX(result.maximumLoan) << "\n";
}

void SaccoSystem::applyForLoan()
{
    Utils::printTitle("Apply for a loan");
    int index = promptMemberIndex();
    if (index < 0)
    {
        return;
    }
    const Member& member = members[index];

    // Quick check first so the officer does not type a whole application for nothing.
    int open = findOpenLoanIndex(member.getMemberId(), "");
    if (open >= 0)
    {
        Utils::printError(member.getFullName() + " already has an unfinished loan " + loans[open].getLoanId() + " (" +
                          loans[open].getStatusName() + "). Only one loan at a time is allowed.");
        return;
    }
    std::cout << "  Member: " << member.getFullName() << "   |   Borrowing limit: "
              << Utils::formatUGX(maximumLoanFor(member)) << "\n";

    double amount = InputHelper::readAmount("  Loan amount (UGX): ", 1, Config::MAX_TRANSACTION);
    int months = InputHelper::readInt("  Repayment period (months): ", 1, 120);
    std::string purpose = readLoanPurpose();

    EligibilityResult result = checkEligibility(member, amount, months, "");
    displayEligibility(result);
    if (!result.eligible)
    {
        Utils::printError("Application NOT registered: the member does not meet the rules marked FAIL.");
        logAudit("Loan application refused for " + member.getMemberId() + " (" + Utils::formatUGX(amount) + ")");
        return;
    }
    displayLoanPreview(amount, settings.getAnnualInterestRate(), months);
    if (!InputHelper::readYesNo("  Submit this loan application?"))
    {
        Utils::printInfo("Application cancelled. Nothing was saved.");
        return;
    }
    Loan loan(settings.generateLoanId(), member.getMemberId(), amount, settings.getAnnualInterestRate(), months,
              purpose, Utils::today(), currentUsername());
    loans.push_back(loan);
    saveAll();
    logAudit("Loan application " + loan.getLoanId() + " for " + member.getMemberId() + " (" + Utils::formatUGX(amount) + ")");
    Utils::printSuccess("Loan application " + loan.getLoanId() + " saved with status PENDING. An Administrator must approve it.");
}

// Prints a table of loans (all of them, or one status only). Returns how many were shown.
int SaccoSystem::listLoans(bool allStatuses, LoanStatus status) const
{
    int count = 0;
    double totalPrincipal = 0.0, totalOutstanding = 0.0;
    for (std::size_t i = 0; i < loans.size(); ++i)
    {
        if (!allStatuses && loans[i].getStatus() != status)
        {
            continue;
        }
        if (count == 0)
        {
            Loan::printTableHeader(std::cout);
        }
        loans[i].printTableRow(std::cout, memberNameOf(loans[i].getMemberId()));
        totalPrincipal += loans[i].getPrincipal();
        totalOutstanding += loans[i].getOutstandingBalance();
        ++count;
    }
    if (count == 0)
    {
        Utils::printInfo("No loans found in this category.");
        return 0;
    }
    std::cout << "  " << std::string(93, '-') << "\n"
              << "  " << std::left << std::setw(36) << ("TOTAL (" + Utils::toString(count) + " loans)") << std::right
              << std::setw(12) << Utils::formatMoney(totalPrincipal) << std::setw(20) << ""
              << std::setw(13) << Utils::formatMoney(totalOutstanding) << "\n";
    return count;
}

void SaccoSystem::updateLoanApplication()
{
    Utils::printTitle("Update a pending loan application");
    int index = promptLoanIndex();
    if (index < 0)
    {
        return;
    }
    Loan& loan = loans[index];
    if (loan.getStatus() != LOAN_PENDING)
    {
        Utils::printError("Only PENDING applications can be changed. Loan " + loan.getLoanId() + " is " +
                          loan.getStatusName() + ".");
        return;
    }
    int memberIndex = findMemberIndex(loan.getMemberId());
    if (memberIndex < 0)
    {
        Utils::printError("The member for this loan no longer exists.");
        return;
    }
    std::cout << "\n";
    loan.displayDetails(std::cout);
    std::cout << "\n";
    Utils::printInfo("Press Enter to keep the value shown in [brackets].");

    double amount = InputHelper::readAmountOrKeep("  Loan amount (UGX)", loan.getPrincipal(), 1, Config::MAX_TRANSACTION);
    int months = InputHelper::readIntOrKeep("  Repayment period (months)", loan.getDurationMonths(), 1, 120);
    std::string purpose = loan.getPurpose();
    if (InputHelper::readYesNo("  Change the loan purpose?"))
    {
        purpose = readLoanPurpose();
    }

    EligibilityResult result = checkEligibility(members[memberIndex], amount, months, loan.getLoanId());
    displayEligibility(result);
    if (!result.eligible)
    {
        Utils::printError("Changes NOT saved: the updated application breaks the rules marked FAIL.");
        return;
    }
    displayLoanPreview(amount, loan.getInterestRate(), months);
    if (!InputHelper::readYesNo("  Save these changes?"))
    {
        Utils::printInfo("Update cancelled. Nothing was saved.");
        return;
    }
    loan.updateApplication(amount, months, purpose);
    saveAll();
    logAudit("Updated loan application " + loan.getLoanId());
    Utils::printSuccess("Loan application " + loan.getLoanId() + " updated.");
}

void SaccoSystem::cancelLoanApplication()
{
    Utils::printTitle("Cancel a loan application");
    int index = promptLoanIndex();
    if (index < 0)
    {
        return;
    }
    const Loan& loan = loans[index];
    if (loan.getStatus() != LOAN_PENDING && loan.getStatus() != LOAN_APPROVED)
    {
        Utils::printError("Only loans not yet paid out (PENDING or APPROVED) can be cancelled. Loan " +
                          loan.getLoanId() + " is " + loan.getStatusName() + ".");
        return;
    }
    std::cout << "\n";
    loan.displayDetails(std::cout);
    std::string loanId = loan.getLoanId();
    if (!InputHelper::readYesNo("  Cancel and delete application " + loanId + "?"))
    {
        Utils::printInfo("Nothing was changed.");
        return;
    }
    loans.erase(loans.begin() + index);
    saveAll();
    logAudit("Cancelled loan application " + loanId);
    Utils::printSuccess("Loan application " + loanId + " has been cancelled and deleted.");
}

// =============================================================================
//  APPROVAL WORKFLOW
// =============================================================================
void SaccoSystem::approveOrRejectLoan()
{
    Utils::printTitle("Approve or reject a loan");
    std::cout << "  Loans waiting for a decision:\n";
    if (listLoans(false, LOAN_PENDING) == 0)
    {
        return;
    }
    std::cout << "\n";
    int index = promptLoanIndex();
    if (index < 0)
    {
        return;
    }
    Loan& loan = loans[index];
    if (loan.getStatus() != LOAN_PENDING)
    {
        Utils::printError("Loan " + loan.getLoanId() + " is " + loan.getStatusName() +
                          ". Only PENDING loans can be approved or rejected.");
        return;
    }
    int memberIndex = findMemberIndex(loan.getMemberId());
    std::cout << "\n  Member Name          : " << memberNameOf(loan.getMemberId()) << "\n";
    loan.displayDetails(std::cout);

    bool eligible = false;
    if (memberIndex >= 0)
    {
        EligibilityResult result = checkEligibility(members[memberIndex], loan.getPrincipal(),
                                                    loan.getDurationMonths(), loan.getLoanId());
        displayEligibility(result);
        eligible = result.eligible;
    }
    if (!eligible)
    {
        Utils::printWarning("The member no longer meets the rules, so this loan can only be rejected.");
    }

    std::cout << "\n  1. Approve\n  2. Reject\n  0. Back (decide later)\n";
    int decision = InputHelper::readMenuChoice(0, 2);
    std::string today = Utils::today();
    switch (decision)
    {
        case 1:
            if (!eligible)
            {
                Utils::printError("Approval blocked: the eligibility rules are not met.");
                return;
            }
            {
                std::string note = InputHelper::readText("  Approval remarks (optional, Enter to skip): ", 0, 60);
                loan.approve(today, currentUsername(), note.empty() ? "Approved" : note);
            }
            saveAll();
            logAudit("Approved loan " + loan.getLoanId());
            Utils::printSuccess("Loan " + loan.getLoanId() + " APPROVED. It can now be paid out (Loan Menu > 8).");
            break;
        case 2:
            {
                std::string reason = InputHelper::readText("  Reason for rejection: ", 3, 60);
                loan.reject(today, currentUsername(), reason);
            }
            saveAll();
            logAudit("Rejected loan " + loan.getLoanId());
            Utils::printSuccess("Loan " + loan.getLoanId() + " REJECTED. The member can apply again later.");
            break;
        default:
            Utils::printInfo("No decision was made.");
            break;
    }
}

void SaccoSystem::recordRepayment()
{
    Utils::printTitle("Record a loan repayment");
    int index = promptLoanIndex();
    if (index < 0)
    {
        return;
    }
    Loan& loan = loans[index];
    if (loan.getStatus() != LOAN_ACTIVE)
    {
        Utils::printError("Repayments can only be made on ACTIVE loans. Loan " + loan.getLoanId() + " is " +
                          loan.getStatusName() + ".");
        return;
    }
    std::string today = Utils::today();
    double outstanding = loan.getOutstandingBalance();
    std::cout << "  Loan ID              : " << loan.getLoanId() << "\n"
              << "  Member               : " << loan.getMemberId() << " - " << memberNameOf(loan.getMemberId()) << "\n"
              << "  Monthly installment  : " << Utils::formatUGX(loan.getInstallment()) << "\n"
              << "  Repaid so far        : " << Utils::formatUGX(loan.getAmountPaid()) << "\n"
              << "  Outstanding balance  : " << Utils::formatUGX(outstanding) << "\n"
              << "  Overdue (arrears)    : " << Utils::formatUGX(loan.getArrears(today)) << "\n"
              << "  Oldest unpaid due on : " << loan.getNextDueDate() << "\n";

    double minimum = (outstanding < Config::MIN_TRANSACTION) ? 1 : Config::MIN_TRANSACTION;
    double amount = InputHelper::readAmount("  Repayment amount (UGX): ", minimum, outstanding);
    if (!InputHelper::readYesNo("  Record repayment of " + Utils::formatUGX(amount) + " on " + loan.getLoanId() + "?"))
    {
        Utils::printInfo("Repayment cancelled.");
        return;
    }
    if (!loan.recordRepayment(amount))
    {
        Utils::printError("The repayment could not be recorded.");
        return;
    }
    recordTransaction(loan.getMemberId(), TXN_LOAN_REPAYMENT, amount, loan.getOutstandingBalance(),
                      "Loan repayment", loan.getLoanId());
    saveAll();
    logAudit("Repayment " + Utils::formatUGX(amount) + " on loan " + loan.getLoanId());

    if (loan.getStatus() == LOAN_COMPLETED)
    {
        Utils::printSuccess("Repayment recorded. Loan " + loan.getLoanId() + " is now FULLY PAID (COMPLETED).");
    }
    else
    {
        Utils::printSuccess("Repayment recorded. Outstanding balance: " + Utils::formatUGX(loan.getOutstandingBalance()));
        double arrears = loan.getArrears(today);
        if (arrears > 0)
        {
            Utils::printWarning("Still overdue: " + Utils::formatUGX(arrears));
        }
    }
}

// Partial, case-insensitive name search: "nak" finds "Nakato Florence".
void SaccoSystem::searchByName()
{
    Utils::printTitle("Search by name");
    std::string text = InputHelper::readText("  Enter full or part of the name: ", 2, 50);
    std::vector<int> matches;
    for (std::size_t i = 0; i < members.size(); ++i)
    {
        if (Utils::containsIgnoreCase(members[i].getFullName(), text))
        {
            matches.push_back(static_cast<int>(i));
        }
    }
    if (matches.empty())
    {
        Utils::printError("No member found whose name contains '" + text + "'.");
        return;
    }
    if (matches.size() == 1)
    {
        Utils::printSuccess("1 record found.");
        showMemberOverview(members[matches[0]]);
        return;
    }
    Utils::printSuccess(Utils::toString(static_cast<int>(matches.size())) + " records found.");
    Member::printTableHeader(std::cout);
    for (std::size_t i = 0; i < matches.size(); ++i)
    {
        members[matches[i]].printTableRow(std::cout);
    }
    std::string pick = Utils::toUpper(Utils::trim(
        InputHelper::readLine("\n  Type a Member ID from the list for details (or press Enter to go back): ")));
    if (!pick.empty())
    {
        int index = findMemberIndex(pick);
        if (index >= 0)
        {
            showMemberOverview(members[index]);
        }
        else
        {
            Utils::printError("No member found with ID '" + pick + "'.");
        }
    }
}

void SaccoSystem::financialSummaryReport() const
{
    Utils::printTitle("SACCO financial summary as at " + Utils::today());
    std::string today = Utils::today();

    double totalSavings = 0.0, totalShares = 0.0;
    for (std::size_t i = 0; i < members.size(); ++i)
    {
        totalSavings += members[i].getSavingsBalance();
        totalShares  += members[i].getShareCapital();
    }

    int statusCount[5] = { 0, 0, 0, 0, 0 };       // indexed by LoanStatus
    double disbursed = 0.0, interestExpected = 0.0, repaid = 0.0;
    double outstanding = 0.0, arrears = 0.0, atRisk = 0.0, pendingValue = 0.0;
    for (std::size_t i = 0; i < loans.size(); ++i)
    {
        const Loan& loan = loans[i];
        ++statusCount[loan.getStatus()];
        if (loan.getStatus() == LOAN_PENDING || loan.getStatus() == LOAN_APPROVED)
        {
            pendingValue += loan.getPrincipal();
        }
        if (loan.getStatus() == LOAN_ACTIVE || loan.getStatus() == LOAN_COMPLETED)
        {
            disbursed        += loan.getPrincipal();
            interestExpected += loan.getTotalInterest();
            repaid           += loan.getAmountPaid();
        }
        if (loan.getStatus() == LOAN_ACTIVE)
        {
            outstanding += loan.getOutstandingBalance();
            double overdue = loan.getArrears(today);
            arrears += overdue;
            if (overdue > 0)
            {
                atRisk += loan.getOutstandingBalance();   // whole balance of a late loan is "at risk"
            }
        }
    }

    double moneyIn = 0.0, moneyOut = 0.0;
    for (std::size_t i = 0; i < transactions.size(); ++i)
    {
        if (transactions[i].isMoneyIn()) moneyIn  += transactions[i].getAmount();
        else                             moneyOut += transactions[i].getAmount();
    }
    double portfolioAtRisk = outstanding > 0 ? atRisk / outstanding * 100.0 : 0.0;

    std::cout << "\n  MEMBERSHIP\n";
    summaryLine("Registered members", Utils::toString(static_cast<int>(members.size())));
    std::cout << "\n  MEMBER FUNDS\n";
    summaryLine("Total savings", Utils::formatUGX(totalSavings));
    summaryLine("Total share capital", Utils::formatUGX(totalShares));
    summaryLine("Total member funds", Utils::formatUGX(totalSavings + totalShares));
    std::cout << "\n  LOAN PORTFOLIO\n";
    summaryLine("Pending applications", Utils::toString(statusCount[LOAN_PENDING]));
    summaryLine("Approved, waiting for payout", Utils::toString(statusCount[LOAN_APPROVED]));
    summaryLine("Value of pending + approved loans", Utils::formatUGX(pendingValue));
    summaryLine("Active loans", Utils::toString(statusCount[LOAN_ACTIVE]));
    summaryLine("Completed loans", Utils::toString(statusCount[LOAN_COMPLETED]));
    summaryLine("Rejected applications", Utils::toString(statusCount[LOAN_REJECTED]));
    summaryLine("Total principal paid out", Utils::formatUGX(disbursed));
    summaryLine("Total interest on loans paid out", Utils::formatUGX(interestExpected));
    summaryLine("Total repayments received", Utils::formatUGX(repaid));
    summaryLine("Outstanding on active loans", Utils::formatUGX(outstanding));
    summaryLine("Amount overdue (arrears)", Utils::formatUGX(arrears));
    summaryLine("Portfolio at risk (PAR)", Utils::toString(portfolioAtRisk, 1) + "%");
    std::cout << "\n  CASH POSITION (from the transaction ledger)\n";
    summaryLine("Money received", Utils::formatUGX(moneyIn));
    summaryLine("Money paid out", Utils::formatUGX(moneyOut));
    summaryLine("Cash that should be at hand / in bank", Utils::formatUGX(moneyIn - moneyOut));
    std::cout << "\n  Money received = deposits + shares + repayments. "
                 "Money paid out = withdrawals + loans + refunds.\n";
}

// Writes a full statement to ANY output stream: the screen (std::cout) or a
// text file (std::ofstream). One function, two uses - no repeated code.
void SaccoSystem::printMemberStatement(const Member& member, std::ostream& out) const
{
    std::string line(92, '=');
    std::string today = Utils::today();
    out << "\n  " << line << "\n"
        << "  " << Config::SACCO_NAME << " - MEMBER STATEMENT\n"
        << "  " << Config::SACCO_ADDRESS << "\n"
        << "  Statement date: " << today << "     Prepared by: " << currentUsername() << "\n"
        << "  " << line << "\n";
    member.displayProfile(out);

    out << "\n  TRANSACTIONS\n";
    Transaction::printLedgerHeader(out);
    int count = 0;
    for (std::size_t i = 0; i < transactions.size(); ++i)
    {
        if (transactions[i].getMemberId() == member.getMemberId())
        {
            transactions[i].printLedgerRow(out);
            ++count;
        }
    }
    if (count == 0)
    {
        out << "  (no transactions)\n";
    }

    out << "\n  LOANS\n";
    double owed = 0.0;
    int loanCount = 0;
    for (std::size_t i = 0; i < loans.size(); ++i)
    {
        const Loan& loan = loans[i];
        if (loan.getMemberId() != member.getMemberId())
        {
            continue;
        }
        ++loanCount;
        owed += loan.getOutstandingBalance();
        out << "  " << loan.getLoanId() << "  " << std::left << std::setw(10) << loan.getStatusName()
            << " Principal " << Utils::formatUGX(loan.getPrincipal())
            << " | Total payable " << Utils::formatUGX(loan.getTotalPayable())
            << " | Repaid " << Utils::formatUGX(loan.getAmountPaid())
            << " | Balance " << Utils::formatUGX(loan.getOutstandingBalance()) << "\n";
        if (loan.getStatus() == LOAN_ACTIVE)
        {
            out << "          Installment " << Utils::formatUGX(loan.getInstallment())
                << " | Oldest unpaid due " << loan.getNextDueDate()
                << " | Overdue " << Utils::formatUGX(loan.getArrears(today)) << "\n";
        }
    }
    if (loanCount == 0)
    {
        out << "  (no loans)\n";
    }

    out << "\n  SUMMARY\n"
        << "  Savings balance         : " << Utils::formatUGX(member.getSavingsBalance()) << "\n"
        << "  Share capital           : " << Utils::formatUGX(member.getShareCapital()) << "\n"
        << "  Loan balance owed       : " << Utils::formatUGX(owed) << "\n"
        << "  Net position            : " << Utils::formatUGX(member.getTotalFunds() - owed)
        << "  (savings + shares - loan balance)\n"
        << "  " << line << "\n";
}

void SaccoSystem::dailyTransactionsReport()
{
    Utils::printTitle("Daily transactions report");
    std::string today = Utils::today();
    std::string date = InputHelper::readDate("  Report date (YYYY-MM-DD)", today, Config::EARLIEST_DATE, today);

    double typeTotals[6] = { 0, 0, 0, 0, 0, 0 };       // indexed by TransactionType
    double moneyIn = 0.0, moneyOut = 0.0;
    int count = 0;
    std::cout << "\n  Transactions on " << date << "\n";
    for (std::size_t i = 0; i < transactions.size(); ++i)
    {
        const Transaction& txn = transactions[i];
        if (txn.getDate() != date)
        {
            continue;
        }
        if (count == 0)
        {
            Transaction::printDailyHeader(std::cout);
        }
        txn.printDailyRow(std::cout);
        typeTotals[txn.getType()] += txn.getAmount();
        if (txn.isMoneyIn()) moneyIn += txn.getAmount();
        else                 moneyOut += txn.getAmount();
        ++count;
    }
    if (count == 0)
    {
        Utils::printInfo("No transactions were recorded on " + date + ".");
        return;
    }
    std::cout << "  " << std::string(90, '-') << "\n"
              << "  Savings deposits     : " << Utils::formatUGX(typeTotals[TXN_SAVINGS_DEPOSIT]) << "\n"
              << "  Share purchases      : " << Utils::formatUGX(typeTotals[TXN_SHARE_PURCHASE]) << "\n"
              << "  Loan repayments      : " << Utils::formatUGX(typeTotals[TXN_LOAN_REPAYMENT]) << "\n"
              << "  Savings withdrawals  : " << Utils::formatUGX(typeTotals[TXN_SAVINGS_WITHDRAWAL]) << "\n"
              << "  Loans paid out       : " << Utils::formatUGX(typeTotals[TXN_LOAN_DISBURSEMENT]) << "\n"
              << "  Account refunds      : " << Utils::formatUGX(typeTotals[TXN_ACCOUNT_CLOSURE]) << "\n"
              << "  ------------------------------------------\n"
              << "  TOTAL MONEY IN       : " << Utils::formatUGX(moneyIn) << "\n"
              << "  TOTAL MONEY OUT      : " << Utils::formatUGX(moneyOut) << "\n"
              << "  NET CASH FOR THE DAY : " << Utils::formatUGX(moneyIn - moneyOut) << "\n"
              << "  Number of entries    : " << count << "\n";
}

void SaccoSystem::exportMemberStatement()
{
    Utils::printTitle("Export member statement to a text file");
    int index = promptMemberIndex();
    if (index < 0)
    {
        return;
    }
    const Member& member = members[index];
    std::string fileName = "statement_" + member.getMemberId() + "_" + Utils::removeChar(Utils::today(), '-') + ".txt";
    std::ofstream file(fileName.c_str());
    if (!file)
    {
        Utils::printError("Could not create the file " + fileName + ".");
        return;
    }
    printMemberStatement(member, file);
    file.close();
    if (!file)
    {
        Utils::printError("Writing the statement file failed.");
        return;
    }
    logAudit("Exported statement for " + member.getMemberId() + " to " + fileName);
    Utils::printSuccess("Statement saved as '" + fileName + "' in the program folder. It can be printed.");
}

// =============================================================================
//  MEMBER SELF-SERVICE (read-only; identity checked with Member ID + NIN)
// =============================================================================
void SaccoSystem::memberSelfService()
{
    showBanner("Member Self-Service");
    Utils::printInfo("Enter your Member ID and your National ID Number (NIN) to view your account.");
    int index = promptMemberIndex();
    if (index < 0)
    {
        InputHelper::pause();
        return;
    }
    std::string nin = Utils::toUpper(Utils::removeChar(Utils::trim(InputHelper::readLine("  Your NIN: ")), ' '));
    if (!Utils::equalsIgnoreCase(members[index].getNationalId(), nin))
    {
        Utils::printError("The NIN does not match our records for this Member ID. Access denied.");
        logAudit("Failed self-service check for " + members[index].getMemberId());
        InputHelper::pause();
        return;
    }
    const std::string memberId = members[index].getMemberId();
    logAudit("Member " + memberId + " opened self-service");

    int choice = 0;
    do
    {
        int current = findMemberIndex(memberId);
        if (current < 0)
        {
            return;
        }
        const Member& member = members[current];
        showBanner("Member Self-Service: " + member.getFullName());
        std::cout << "  1. My Account Summary\n"
                  << "  2. My Statement\n"
                  << "  3. My Loans and Repayment Schedule\n"
                  << "  4. How Much Can I Borrow?\n"
                  << "  0. Exit Self-Service\n";
        choice = InputHelper::readMenuChoice(0, 4);
        switch (choice)
        {
            case 1:
                showMemberOverview(member);
                break;
            case 2:
                printMemberStatement(member, std::cout);
                break;
            case 3:
            {
                int shown = 0;
                for (std::size_t i = 0; i < loans.size(); ++i)
                {
                    if (loans[i].getMemberId() == memberId)
                    {
                        Utils::printTitle("Loan " + loans[i].getLoanId());
                        loans[i].displayDetails(std::cout);
                        std::cout << "\n";
                        loans[i].displaySchedule(std::cout);
                        ++shown;
                    }
                }
                if (shown == 0)
                {
                    Utils::printInfo("You have no loans on record.");
                }
                break;
            }
            case 4:
            {
                double limit = maximumLoanFor(member);
                std::cout << "\n  You may borrow up to " << Utils::formatUGX(limit) << " over up to "
                          << settings.getMaximumDurationMonths() << " months, if every rule below passes:\n";
                displayEligibility(checkEligibility(member, limit, settings.getMaximumDurationMonths(), ""));
                Utils::printInfo("Visit the SACCO office to apply. A loan officer will guide you.");
                break;
            }
            default:
                break;
        }
        if (choice != 0)
        {
            InputHelper::pause();
        }
    } while (choice != 0);
}
